-- ==================== DRAGGING SYSTEM ====================
local function attachDragger(dragHandle, targetWindow)
   local dragging = false
   local dragStart = nil
   local startPos = nil
   dragHandle.InputBegan:Connect(function(input)
       if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
           dragging = true
           dragStart = input.Position
           startPos = targetWindow.Position
           input.Changed:Connect(function()
               if input.UserInputState == Enum.UserInputState.End then
                   dragging = false
               end
           end)
       end
   end)
   UserInputService.InputChanged:Connect(function(input)
       if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
           local delta = input.Position - dragStart
           targetWindow.Position = UDim2.new(
               startPos.X.Scale,
               startPos.X.Offset + delta.X,
               startPos.Y.Scale,
               startPos.Y.Offset + delta.Y
           )
       end
   end)
end

local function setActive(val)
   if state.Active == val then return end
   state.Active = val
   if val then
       state.StartedAt = os.clock()
       startBypass(state.Power, state.Version)
   else
       stopBypass()
   end
   updateVisual()
end

-- ==================== AUTO BRAINROT ENGINE ====================
local function checkBrainrot()
   local char = LocalPlayer.Character
   if char then
       local hum = char:FindFirstChildOfClass("Humanoid")
       if hum and hum.WalkSpeed < 25 then
           return true
       end
   end
   local pGui = LocalPlayer:FindFirstChild("PlayerGui")
   if pGui then
       local targets = {
           "DuelsMachineSession",
           "DuelsMachine",
           "BrainrotUI",
           "BrainrotSession",
           "SessionGUI",
           "DuelsGUI"
       }
       for _, name in ipairs(targets) do
           if pGui:FindFirstChild(name) then
               return true
           end
       end
   end
   return false
end
RunService.Heartbeat:Connect(function()
   if not state.AutoBrainrot then return end
   local isBrainrot = checkBrainrot()
   if isBrainrot and not state.LastBrainrotState then
       state.LastBrainrotState = true
       setActive(true)
   elseif not isBrainrot and state.LastBrainrotState then
       state.LastBrainrotState = false
       setActive(false)
   end
end)

       local mins = math.floor(elapsed / 60)
       local secs = elapsed % 60
       StatusText.Text = string.format("PREVIEW [%02d:%02d] - %d PWR", mins, secs, state.Power)
   end
end)
